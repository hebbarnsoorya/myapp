// ---------- Layout Component ----------

      { label: "Analytics", icon: <BarChart3 size={18} />, href: "#analytics" },

  useEffect(() => {
    const tests: { name: string; pass: boolean; details?: string }[] = [];
    try {
      // Original tests
      tests.push({ name: "activity has 12 points", pass: activity.length === 12 });
      tests.push({ name: "tableData has 7 rows", pass: tableData.length === 7 });
      const brand1 = getComputedStyle(document.documentElement).getPropertyValue("--brand-1").trim();

      const ringElement = React.createElement(RingIcon as any, { className: "h-3.5 w-3.5" });
      tests.push({ name: "Ring icon is renderable", pass: React.isValidElement(ringElement) });

      // Additional tests
      tests.push({ name: "nav has 6 items", pass: nav.length === 6 });
      tests.push({ name: "chart keys present", pass: Object.keys(activity[0]).sort().join(",") === ["name","revenue","users"].join(",") });
      const brand2 = getComputedStyle(document.documentElement).getPropertyValue("--brand-2").trim();
      tests.push({ name: "CSS var --brand-2 is set", pass: !!brand2, details: brand2 });
    } catch (e) {
      console.error("Self-tests encountered an error:", e);
    } finally {
      const allPass = tests.every((t) => t.pass);
      console.group("AdminLayout self-tests");
      for (const t of tests) console[t.pass ? "log" : "error"](`${t.pass ? "PASS" : "FAIL"} - ${t.name}${t.details ? ` (${t.details})` : ""}`);
      console.groupEnd();
      if (!allPass) console.warn("One or more self-tests failed. Check details above.");

              <div className="text-sm font-semibold tracking-tight text-zinc-800 dark:text-zinc-100">{org}</div>
              <div className="text-xs text-zinc-500">Organization</div>
            </div>
          </div>

          {/* Middle cluster: primary nav & search */}
          <div className="ml-2 flex-1 flex items-center gap-2">
            <nav className="hidden md:flex items-center gap-1">
              {/* Mixed: some dropdowns, some direct links */}
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="ghost" className="gap-1">
                    Products <ChevronDown className="h-4 w-4" />
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent align="start">
                  <DropdownMenuLabel>Products</DropdownMenuLabel>
                  <DropdownMenuSeparator />
                  <DropdownMenuItem>New</DropdownMenuItem>
                  <DropdownMenuItem>Browse</DropdownMenuItem>
                  <DropdownMenuItem>Categories</DropdownMenuItem>
                </DropdownMenuContent>
              </DropdownMenu>
              {nav.slice(0, 2).map((item) => (
                <Button key={item.label} variant="ghost" className="gap-2">
                  {item.icon}
                  {item.label}
                </Button>
              ))}
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="ghost" className="gap-1">
                    More <ChevronDown className="h-4 w-4" />
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent>
                  {nav.slice(2).map((item) => (
                    <DropdownMenuItem key={item.label} className="gap-2">
                      {item.icon}
                      {item.label}
                    </DropdownMenuItem>

              </DropdownMenu>
            </nav>

              <div className="relative w-full">

            </div>
          </div>

          {/* Right cluster: notifications, profile, switches */}
          <div className="flex items-center gap-1 sm:gap-2">
            <IconPill icon={<Bell className="h-4 w-4" />} label="9" />

            {nav.slice(0, 5).map((item) => (

              <CardContent className="h-72">

            </Card>

            <Card className="rounded-2xl">
              <CardHeader>
                <CardTitle>Quarterly Breakdown</CardTitle>
              </CardHeader>
              <CardContent className="h-72">

            <CardHeader>

                <CardTitle>Projects</CardTitle>
                <div className="flex items-center gap-2">

                  <thead>
                    <tr className="text-left text-zinc-500">
                      <th className="py-2 pr-4">#</th>
                      <th className="py-2 pr-4">Project</th>
                      <th className="py-2 pr-4">Owner</th>

                            className="inline-flex items-center rounded-full px-2 py-0.5 text-xs bg-[var(--brand-1)]/10 text-zinc-700 dark:text-zinc-200"

                            {row.status}
                          </span>
                        </td>
                        <td className="py-3 pr-4">
                          <div className="h-2 w-40 bg-zinc-200 dark:bg-zinc-800 rounded-full overflow-hidden">
                            <div
                              className="h-full bg-[var(--brand-1)]"

                    ))}

          </Card>

          {/* Footer */}

// ---------- Reusable UI bits ----------
function IconPill({ icon, label }: { icon: React.ReactNode; label?: string }) {
  return (
    <div className="relative">
      <Button variant="ghost" size="icon" className="h-9 w-9">
        {icon}
      </Button>
      {label && (
        <span className="absolute -top-1 -right-1 min-w-5 h-5 rounded-full bg-[var(--brand-1)] text-white text-[10px] grid place-items-center px-1">
          {label}
        </span>
      )}
    </div>
  );
}

function LabeledSelect({ icon, value, setValue, label, children }: any) {
  return (
    <div className="flex items-center gap-1">
      <span className="text-zinc-500">{icon}</span>
      <Select value={value} onValueChange={(v) => setValue(v)}>
        <SelectTrigger className="h-9 w-[8.5rem] rounded-xl">
          <SelectValue placeholder={label} />
        </SelectTrigger>
        <SelectContent className="rounded-xl">
          {children}
        </SelectContent>
      </Select>
    </div>
  );
}

function KpiCard({ title, value, sub }: { title: string; value: string; sub: string }) {
  return (
    <Card className="rounded-2xl border-transparent bg-[rgb(var(--surface))] shadow-sm">
      <CardHeader className="pb-2">
        <CardTitle className="text-sm text-zinc-500">{title}</CardTitle>
      </CardHeader>
      <CardContent>
        <div className="text-2xl font-semibold tracking-tight">{value}</div>
        <div className="text-xs text-zinc-500 mt-1">{sub}</div>
      </CardContent>
    </Card>
  );
}
